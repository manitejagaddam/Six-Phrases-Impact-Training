class stack
{
   Node [] safe;
   int top;
stack ( int count)
{
  safe = new Node [ count ]
top = -1;
}

push ( Node keep )
{
   top++;
   safe [ top ] = keep;
}
Node pop()
{
   return safe[top--];

}
}
